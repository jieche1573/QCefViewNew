增加记录

```JSON
  qtClient.queryFlyAddReq(jsonStr);
  qtClient.queryFlyAdd.connect(function (newJsonStr) {
    alert(newJsonStr)
  });
```

删除记录

```JSON
  qtClient.queryFlyDelReq("123334");
  qtClient.queryFlyDel.connect(function (ok) {
    alert(ok)
  });
```

编辑记录

```JSON
  qtClient.queryFlyEditReq(jsonStr);
  qtClient.updateRecord.connect(function (ok) {
    alert(ok)
  });
```

查询记录（模糊搜索）

```JSON
  qtClient.queryFlySearchReq(Keyword);
  qtClient.queryFlySearch.connect(function (jsonStr) {
    alert(jsonStr)
  });
```

根据ID查询单条记录

```JSON
  qtClient.queryFlyByIdReq("20251109144639678678cc1ae7");
  qtClient.queryFlyById.connect(function (jsonStr) {
    alert(jsonStr)
  });
```
